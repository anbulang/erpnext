# ERPNext 食品原料批发系统 - 3 天快速上线方案

## 🎯 食品原料批发 vs 零售门店的区别

| 维度 | 零售门店 | 食品原料批发 |
|------|----------|--------------|
| 客户 | 散客（B2C） | 餐厅、面包店（B2B） |
| 交易 | 当场收款 | 月结账期（30-60天） |
| 数量 | 小额高频 | 大宗订单 |
| 定价 | 固定零售价 | 客户专属价 |
| 配送 | 自提 | 送货上门 |
| 核心单据 | POS Invoice | 报价单 → 销售订单 → 送货单 → 发票 |
| 关键功能 | 收银结账 | 批次管理、信用额度、账期 |

**结论：不需要 POS 模块，核心是 Selling + Stock + Accounts**

```python
def set_customer_prices():
    """批量设置客户专属价格"""
    
    # 获取所有商品和基础价格
    items = frappe.get_all("Item", 
                          filters={"is_sales_item": 1},
                          fields=["name", "standard_rate"])
    
    # 价格策略
    price_rules = {
        "餐饮客户价": 0.95,  # 95折
        "烘焙客户价": 0.92,  # 92折
        "VIP 客户价": 0.88   # 88折
    }
    
    for price_list, discount in price_rules.items():
        for item in items:
            # 检查是否已有价格
            if frappe.db.exists("Item Price", {
                "item_code": item.name,
                "price_list": price_list
            }):
                continue
            
            price = frappe.new_doc("Item Price")
            price.item_code = item.name
            price.price_list = price_list
            price.price_list_rate = item.standard_rate * discount
            price.selling = 1
            price.insert()
        
        print(f"✓ 设置 {price_list}: {len(items)} 个商品")
    
    frappe.db.commit()
    print("✅ 价格设置完成")
```

```python
def import_opening_stock(excel_file):
    """导入期初库存"""
    df = pd.read_excel(excel_file)
    
    recon = frappe.new_doc("Stock Reconciliation")
    recon.company = "你的公司"
    recon.posting_date = frappe.utils.today()
    recon.posting_time = frappe.utils.nowtime()
    recon.purpose = "Opening Stock"
    
    for _, row in df.iterrows():
        # 先创建批次
        if pd.notna(row.get('batch_no')):
            if not frappe.db.exists("Batch", row['batch_no']):
                batch = frappe.new_doc("Batch")
                batch.batch_id = row['batch_no']
                batch.item = row['item_code']
                if pd.notna(row.get('expiry_date')):
                    batch.expiry_date = row['expiry_date']
                batch.insert()
        
        # 添加库存
        recon.append("items", {
            "item_code": row['item_code'],
            "warehouse": row['warehouse'],
            "qty": row['qty'],
            "valuation_rate": row['valuation_rate'],
            "batch_no": row.get('batch_no')
        })
    
    recon.insert()
    recon.submit()
    
    print(f"✅ 导入期初库存: {recon.name}")
    print(f"   商品数: {len(recon.items)}")
    return recon.name
```

```python
def sales_order_to_delivery_note(so_name):
    """销售订单转送货单"""
    from frappe.model.mapper import get_mapped_doc
    
    dn = get_mapped_doc("Sales Order", so_name, {
        "Sales Order": {
            "doctype": "Delivery Note"
        },
        "Sales Order Item": {
            "doctype": "Delivery Note Item"
        }
    })
    
    # 设置批次（重要！食品必须追溯）
    for item in dn.items:
        # 自动选择最早过期的批次（FEFO - First Expire First Out）
        batch = frappe.db.sql("""
            SELECT batch_id, expiry_date, qty
            FROM `tabBatch`
            WHERE item = %(item)s
              AND qty > 0
            ORDER BY expiry_date ASC
            LIMIT 1
        """, {"item": item.item_code}, as_dict=1)
        
        if batch:
            item.batch_no = batch[0].batch_id
    
    dn.insert()
    dn.submit()  # 提交后自动扣减库存
    
    print(f"✅ 创建送货单: {dn.name}")
    return dn.name
```

```python
def monthly_billing(customer, month):
    """月底对账开票
    
    Args:
        customer: 客户名称
        month: 月份 "2026-06"
    """
    from frappe.utils import getdate, get_first_day, get_last_day
    
    start_date = get_first_day(month)
    end_date = get_last_day(month)
    
    # 查找本月所有已送货未开票的送货单
    dns = frappe.get_all("Delivery Note",
                        filters={
                            "customer": customer,
                            "docstatus": 1,
                            "posting_date": ["between", [start_date, end_date]],
                            "status": ["!=", "Completed"]  # 未完全开票
                        },
                        fields=["name", "grand_total"])
    
    if not dns:
        print(f"⚠️  {customer} 在 {month} 无待开票送货单")
        return
    
    print(f"\n📋 {customer} {month} 月对账单")
    print("=" * 60)
    print(f"{'送货单号':<20} {'金额':>15}")
    print("-" * 60)
    
    total = 0
    for dn in dns:
        print(f"{dn.name:<20} ¥{dn.grand_total:>14.2f}")
        total += dn.grand_total
    
    print("-" * 60)
    print(f"{'合计':<20} ¥{total:>14.2f}")
    print("=" * 60)
    
    # 生成销售发票（汇总）
    confirm = input(f"\n是否开具发票？(y/n): ")
    if confirm.lower() != 'y':
        return
    
    invoice = frappe.new_doc("Sales Invoice")
    invoice.customer = customer
    invoice.posting_date = end_date
    invoice.due_date = frappe.utils.add_days(end_date, 30)  # 30 天账期
    
    # 合并所有送货单
    for dn in dns:
        dn_doc = frappe.get_doc("Delivery Note", dn.name)
        for item in dn_doc.items:
            invoice.append("items", {
                "item_code": item.item_code,
                "item_name": item.item_name,
                "qty": item.qty,
                "rate": item.rate,
                "warehouse": item.warehouse,
                "delivery_note": dn.name,
                "dn_detail": item.name
            })
    
    invoice.insert()
    invoice.submit()
    
    print(f"\n✅ 开票成功: {invoice.name}")
    print(f"   开票金额: ¥{invoice.grand_total}")
    print(f"   到期日: {invoice.due_date}")
    return invoice.name
```

```python
def get_expiring_items(days=30):
    """临期商品预警"""
    
    from frappe.utils import add_days, today
    
    alert_date = add_days(today(), days)
    
    result = frappe.db.sql("""
        SELECT 
            b.item,
            i.item_name,
            b.batch_id,
            b.expiry_date,
            DATEDIFF(b.expiry_date, CURDATE()) as days_to_expire,
            SUM(sle.actual_qty) as qty
        FROM 
            `tabBatch` b
        INNER JOIN 
            `tabItem` i ON b.item = i.name
        INNER JOIN
            `tabStock Ledger Entry` sle ON sle.batch_no = b.batch_id
        WHERE 
            b.expiry_date <= %(alert_date)s
            AND b.expiry_date >= CURDATE()
        GROUP BY 
            b.batch_id
        HAVING 
            qty > 0
        ORDER BY 
            b.expiry_date ASC
    """, {"alert_date": alert_date}, as_dict=1)
    
    print(f"\n⚠️  临期商品预警（{days} 天内过期）")
    print("=" * 80)
    print(f"{'商品':<20} {'批次号':<15} {'过期日期':<12} {'剩余天数':>8} {'库存':>8}")
    print("-" * 80)
    
    for item in result:
        print(f"{item.item_name:<20} {item.batch_id:<15} {item.expiry_date} "
              f"{item.days_to_expire:>8} {item.qty:>8.0f}")
    
    return result
```

```python
def get_monthly_sales_summary(month="2026-06"):
    """月度销售汇总"""
    
    from frappe.utils import get_first_day, get_last_day
    
    start = get_first_day(month)
    end = get_last_day(month)
    
    result = frappe.db.sql("""
        SELECT 
            si.customer,
            COUNT(DISTINCT si.name) as invoice_count,
            SUM(si.grand_total) as total_amount,
            SUM(si.outstanding_amount) as outstanding
        FROM 
            `tabSales Invoice` si
        WHERE 
            si.docstatus = 1
            AND si.posting_date BETWEEN %(start)s AND %(end)s
        GROUP BY 
            si.customer
        ORDER BY 
            total_amount DESC
    """, {"start": start, "end": end}, as_dict=1)
    
    print(f"\n📊 {month} 销售汇总")
    print("=" * 80)
    print(f"{'客户':<25} {'开票数':>8} {'销售额':>15} {'未收款':>15}")
    print("-" * 80)
    
    total_sales = 0
    total_outstanding = 0
    
    for row in result:
        print(f"{row.customer:<25} {row.invoice_count:>8} "
              f"¥{row.total_amount:>14.2f} ¥{row.outstanding:>14.2f}")
        total_sales += row.total_amount
        total_outstanding += row.outstanding
    
    print("-" * 80)
    print(f"{'合计':<25} {'':<8} ¥{total_sales:>14.2f} ¥{total_outstanding:>14.2f}")
    
    return result
```

## ⚠️ 常见问题

### Q1: 如何处理客户退货？
A: 创建 Sales Return (退货)
- 打开原送货单 → Make → Return
- 选择退货商品和数量
- 提交后自动生成红字发票

### Q2: 如何处理多计量单位？（如：箱/包/公斤）
A: 使用 UOM Conversion
- Item → Unit of Measure Conversion
- 设置换算关系：1 箱 = 10 包 = 250 公斤

### Q3: 如何设置自动库存预警？
A: 在 Item 中设置 Reorder Level
- Item → Reorder → Add
- 设置最低库存和重订货量
- 系统会自动生成 Material Request
