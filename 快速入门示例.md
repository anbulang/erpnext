## 示例 1：创建自定义字段

### 场景
为 Customer（客户）DocType 添加一个"VIP 等级"字段

### 方法一：通过界面（推荐初学者）
```
1. 登录 ERPNext
2. 搜索 "Custom Field"
3. 点击 "New"
4. 填写：
   - DocType: Customer
   - Label: VIP Level
   - Field Type: Select
   - Options: Bronze\nSilver\nGold\nPlatinum
5. 保存
```

### 方法二：通过代码
```python
import frappe

def create_custom_field():
    """创建自定义字段"""
    if not frappe.db.exists("Custom Field", "Customer-vip_level"):
        frappe.get_doc({
            "doctype": "Custom Field",
            "dt": "Customer",
            "label": "VIP Level",
            "fieldname": "vip_level",
            "fieldtype": "Select",
            "options": "Bronze\nSilver\nGold\nPlatinum",
            "insert_after": "customer_name"
        }).insert()
        print("✅ 自定义字段创建成功")

# 在 bench console 中执行
# bench --site mysite.localhost console
# >>> from path.to.file import create_custom_field
# >>> create_custom_field()
```

## 示例 2：添加自定义验证逻辑

### 场景
销售订单金额超过 10000 时需要经理审批

### Server Script 方式
```python
# 在 ERPNext 界面创建 Server Script
# DocType: Sales Order
# Event: Before Save

import frappe

def validate():
    if frappe.form_dict.grand_total > 10000:
        if not frappe.form_dict.approval_status == "Approved":
            frappe.throw("订单金额超过 10000，需要经理审批！")

validate()
```

### 自定义应用方式
```python
# 在 hooks.py 中添加
doc_events = {
    "Sales Order": {
        "validate": "my_custom_app.custom.validate_sales_order"
    }
}

# 创建文件：my_custom_app/custom.py
import frappe
from frappe import _

def validate_sales_order(doc, method):
    """验证销售订单"""
    if doc.grand_total > 10000:
        if not doc.get("approval_status") == "Approved":
            frappe.throw(
                _("订单金额 {0} 超过限额，需要经理审批！").format(doc.grand_total),
                frappe.ValidationError
            )
```

## 示例 3：创建自定义报表

### 场景
统计每个客户的年度购买金额

### 步骤
```bash

```

### 创建报表 Python 文件
```python
# customer_purchase_analysis.py

import frappe
from frappe import _

def execute(filters=None):
    """执行报表查询"""
    columns = get_columns()
    data = get_data(filters)
    return columns, data

def get_columns():
    """定义报表列"""
    return [
        {
            "fieldname": "customer",
            "label": _("客户"),
            "fieldtype": "Link",
            "options": "Customer",
            "width": 200
        },
        {
            "fieldname": "customer_name",
            "label": _("客户名称"),
            "fieldtype": "Data",
            "width": 200
        },
        {
            "fieldname": "total_amount",
            "label": _("总购买金额"),
            "fieldtype": "Currency",
            "width": 150
        },
        {
            "fieldname": "invoice_count",
            "label": _("发票数量"),
            "fieldtype": "Int",
            "width": 100
        }
    ]

def get_data(filters):
    """获取数据"""
    conditions = get_conditions(filters)

    data = frappe.db.sql(f"""
        SELECT
            si.customer,
            si.customer_name,
            SUM(si.grand_total) as total_amount,
            COUNT(si.name) as invoice_count
        FROM
            `tabSales Invoice` si
        WHERE
            si.docstatus = 1
            {conditions}
        GROUP BY
            si.customer
        ORDER BY
            total_amount DESC
    """, filters, as_dict=1)

    return data

def get_conditions(filters):
    """构建查询条件"""
    conditions = ""

    if filters.get("from_date"):
        conditions += " AND si.posting_date >= %(from_date)s"

    if filters.get("to_date"):
        conditions += " AND si.posting_date <= %(to_date)s"

    if filters.get("customer"):
        conditions += " AND si.customer = %(customer)s"

    return conditions
```

## 示例 4：创建自定义 API

### 场景
创建一个 API 获取客户余额

```python
# 在自定义应用中创建：my_custom_app/api.py

import frappe
from frappe import _

@frappe.whitelist()
def get_customer_balance(customer):
    """
    获取客户余额

    参数:
        customer: 客户ID

    返回:
        dict: {
            "customer": "客户ID",
            "customer_name": "客户名称",
            "outstanding_amount": 余额
        }
    """
    # 检查权限
    if not frappe.has_permission("Customer", "read"):
        frappe.throw(_("没有权限访问客户信息"), frappe.PermissionError)

    # 验证客户存在
    if not frappe.db.exists("Customer", customer):
        frappe.throw(_("客户 {0} 不存在").format(customer))

    # 查询余额
    customer_doc = frappe.get_doc("Customer", customer)

    return {
        "customer": customer,
        "customer_name": customer_doc.customer_name,
        "outstanding_amount": frappe.db.get_value(
            "Customer",
            customer,
            "outstanding_amount"
        ) or 0
    }

@frappe.whitelist()
def create_sales_order(customer, items):
    """
    创建销售订单

    参数:
        customer: 客户ID
        items: 商品列表 [{"item_code": "ITEM-001", "qty": 5}]

    返回:
        str: 销售订单名称
    """
    import json

    # 解析 JSON 参数
    if isinstance(items, str):
        items = json.loads(items)

    # 创建销售订单
    so = frappe.get_doc({
        "doctype": "Sales Order",
        "customer": customer,
        "delivery_date": frappe.utils.add_days(frappe.utils.today(), 7),
        "items": []
    })

    # 添加商品
    for item in items:
        so.append("items", {
            "item_code": item.get("item_code"),
            "qty": item.get("qty", 1),
            "delivery_date": frappe.utils.add_days(frappe.utils.today(), 7)
        })

    # 保存并提交
    so.insert()
    so.submit()

    return so.name
```

## 示例 5：添加前端自定义按钮

### 场景
在销售发票中添加"发送提醒"按钮

```javascript
// 创建文件：my_custom_app/public/js/sales_invoice.js

frappe.ui.form.on('Sales Invoice', {
    refresh: function(frm) {
        // 只在已提交状态显示按钮
        if (frm.doc.docstatus === 1) {
            frm.add_custom_button(__('发送付款提醒'), function() {
                send_payment_reminder(frm);
            }, __('操作'));
        }
    }
});

function send_payment_reminder(frm) {
    frappe.confirm(
        __('确定要向客户 {0} 发送付款提醒吗？', [frm.doc.customer_name]),
        function() {
            // 确认后调用后端方法
            frappe.call({
                method: 'my_custom_app.api.send_payment_reminder',
                args: {
                    sales_invoice: frm.doc.name
                },
                callback: function(r) {
                    if (r.message) {
                        frappe.msgprint(__('提醒已发送'));
                    }
                }
            });
        }
    );
}
```

## 示例 6：创建后台任务

### 场景
每天自动发送未付款发票提醒

```python
# 在 hooks.py 中添加定时任务
scheduler_events = {
    "daily": [
        "my_custom_app.tasks.send_overdue_invoice_reminders"
    ]
}

# 创建任务文件：my_custom_app/tasks.py
import frappe
from frappe.utils import today, add_days

def send_overdue_invoice_reminders():
    """发送逾期发票提醒"""
    # 获取所有逾期未付款的发票
    overdue_invoices = frappe.get_all(
        "Sales Invoice",
        filters={
            "docstatus": 1,
            "status": ["!=", "Paid"],
            "due_date": ["<", today()]
        },
        fields=["name", "customer", "customer_name", "grand_total", "outstanding_amount", "due_date"]
    )

    for invoice in overdue_invoices:
        # 获取客户邮箱
        customer_email = frappe.db.get_value("Customer", invoice.customer, "email_id")

        if customer_email:
            # 发送邮件
            frappe.sendmail(
                recipients=[customer_email],
                subject=f"付款提醒 - 发票 {invoice.name}",
                message=f"""
                尊敬的 {invoice.customer_name}，

                您的发票 {invoice.name} 已逾期。

                发票金额: {invoice.grand_total}
                未付金额: {invoice.outstanding_amount}
                到期日期: {invoice.due_date}

                请尽快安排付款。

                谢谢！
                """
            )

            # 记录日志
            frappe.log_error(
                f"已向 {customer_email} 发送提醒",
                f"发票提醒 - {invoice.name}"
            )

    return f"已处理 {len(overdue_invoices)} 张逾期发票"
```

### 手动触发后台任务
```python

send_overdue_invoice_reminders()
```

## 示例 7：数据库查询优化

### ❌ 不好的做法（N+1 查询问题）
```python
# 获取所有销售订单
orders = frappe.get_all("Sales Order", fields=["name"])

# 对每个订单查询客户信息（会执行 N 次查询）
for order in orders:
    customer = frappe.get_doc("Customer", order.customer)
    print(customer.customer_name)
```

### ✅ 好的做法（一次性查询）
```python
# 使用 JOIN 一次性获取所有数据
orders = frappe.db.sql("""
    SELECT
        so.name,
        so.customer,
        c.customer_name,
        c.customer_group
    FROM
        `tabSales Order` so
    LEFT JOIN
        `tabCustomer` c ON so.customer = c.name
    WHERE
        so.docstatus = 1
""", as_dict=1)

for order in orders:
    print(order.customer_name)
```

### 使用 ORM 的优化方法
```python
# 使用 pluck 只获取需要的字段
customer_names = frappe.db.get_all(
    "Customer",
    filters={"disabled": 0},
    pluck="customer_name"
)

# 使用 limit 限制查询数量
recent_orders = frappe.get_all(
    "Sales Order",
    filters={"docstatus": 1},
    fields=["name", "customer", "grand_total"],
    order_by="creation desc",
    limit=100
)
```

@frappe.whitelist()
def upload_file():
    """处理文件上传"""
    file = frappe.request.files.get('file')

    if file:
        # 保存文件
        ret = frappe.get_doc({
            "doctype": "File",
            "file_name": file.filename,
            "is_private": 1,
            "content": file.stream.read()
        })
        ret.save()

        return ret.file_url
